async (page) => {
 const checks=[];const ok=(v,n)=>{if(!v)throw new Error(n);checks.push(n);};
 await page.setViewportSize({width:1440,height:1000});
 await page.goto('http://127.0.0.1:33324/login');
 await page.getByRole('heading',{name:'登录忆程',exact:true}).waitFor();
 await page.evaluate(()=>document.fonts.ready);
 await page.screenshot({path:'evidence/pre-r4/20261023-home-story/after-login-desktop.png'});
 await page.getByLabel('账号',{exact:true}).fill('browser-acceptance-20261023');
 await page.getByLabel('密码',{exact:true}).fill('browser-local-password');
 await page.getByRole('button',{name:'显示密码',exact:true}).click();
 ok(await page.getByLabel('密码',{exact:true}).getAttribute('type')==='text','password visibility toggle');
 await page.getByRole('button',{name:'隐藏密码',exact:true}).click();
 await page.getByRole('button',{name:'登录工作台',exact:true}).click();
 await page.locator('#auth-error').waitFor();
 ok((await page.locator('#auth-error').innerText()).includes('请先验证注册邮箱'),'unverified account cannot log in');
 await page.goto('http://127.0.0.1:33324/verify-email?token=VERIFY_TOKEN');
 await page.getByRole('button',{name:'验证我的邮箱',exact:true}).click();
 await page.getByRole('heading',{name:'操作已完成',exact:true}).waitFor();
 ok(!page.url().includes('token='),'token removed from browser address');
 await page.goto('http://127.0.0.1:33324/login?from=%2Fprojects');
 await page.getByLabel('账号',{exact:true}).fill('browser-acceptance-20261023');
 await page.getByLabel('密码',{exact:true}).fill('browser-local-password');
 await page.getByRole('button',{name:'登录工作台',exact:true}).click();
 await page.waitForURL('**/projects');
 ok(page.url().endsWith('/projects'),'verified login respects workspace return path');
 const list=await (await page.request.get('http://127.0.0.1:33324/api/projects')).json();ok(list.projects.length===0,'new browser account sees no seeded projects');
 const response=await page.request.post('http://127.0.0.1:33324/api/projects',{data:{title:'注册验收独立项目',description:'只用于本地隔离账号验收的项目资料，不涉及个人真实项目。',goal:'验证账号能管理自己的项目',scenario:'COURSE_DESIGN'}});ok(response.status()===201,'new account can create own project');
 const created=(await response.json()).project;
 const capture=await page.request.post('http://127.0.0.1:33324/api/projects/'+created.id+'/captures',{data:{rawText:'验收记录：已完成账号和项目归属验证，下一步核对来源。',sourceType:'TEXT'}});ok(capture.status()===201,'new account can save project record in rule mode');
 await page.getByRole('button',{name:'browser-acceptance-20261023',exact:true}).click();
 await page.getByRole('button',{name:'退出登录',exact:true}).click();
 await page.waitForURL('**/login');
 ok((await page.request.get('http://127.0.0.1:33324/api/auth/me')).status()===401,'UI logout revokes server session');
 await page.goto('http://127.0.0.1:33324/forgot-password');
 await page.getByLabel('邮箱',{exact:true}).fill('browser-acceptance@example.test');
 await page.getByRole('button',{name:'发送重置邮件',exact:true}).click();
 await page.getByRole('heading',{name:'操作已完成',exact:true}).waitFor();
 ok(true,'password recovery submitted through actual backend and TLS SMTP');
 return {checks,source:'isolated E2E database and local TLS SMTP; no Resend or live users'};
}
